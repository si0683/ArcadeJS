import Arcade from 'arcadejs';
import 'arcadejs/arcade.css';
Arcade.create({
  title: 'Fruit Drop', emoji: '🍉', world: 'sky',
  lives: 3, buttons: ['left', 'right'],
  setup(game) {
    game.data.hero = game.add({ emoji: '🧺', x: 400, y: 540, size: 70, inside: true });
    game.every(0.7, g => g.add({ emoji: g.pick(['🍎', '🍌']),
      x: g.rand(40, 760), y: -30, vy: 220, tag: 'fruit', removeOffscreen: true }));
  },
  update(game) {
    const hero = game.data.hero;
    hero.vx = game.key.left ? -500 : game.key.right ? 500 : 0;
    const fruit = game.hitAny(hero, 'fruit');
    if (fruit) { fruit.remove(); game.addScore(1, fruit.x, fruit.y); game.sound('coin'); }
  }
});
Features

Everything around your game, done

Screens & input

Menu, settings, countdown, pause, game over, high score. Keyboard, mouse, touch, swipes, on-screen buttons.

Objects & camera

Emoji, sprites and shapes with velocity, gravity, collisions, platforms and a follow camera.

Sound & music

99 sounds (Kenney, ZzFX 8-bit, built-in) and 5 background tracks.

Sprites

201 Kenney sprites; characters animate walking and jumping by themselves.

Physics

physics: true — Matter.js bodies, ropes, collision events, drag-and-throw.

3D

mode: '3d' — Three.js boxes, spheres, emoji billboards, chase camera.

Progression

Levels, lives, upgrade picker (game.choose) and quizzes (game.ask).

i18n

English and Russian interface, every text overridable.

TypeScript

Typed API, ES module, safe SSR import for Next.js and friends.
